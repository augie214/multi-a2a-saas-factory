  isAnomaly(value, metricDef) {
    const direction = metricDef.direction || 'higher-is-better';
    
    if (direction === 'lower-is-better') {
      return value > metricDef.thresholds.warning;
    } else {
      return value < metricDef.thresholds.warning;
    }
  }
  
  getRelevantThreshold(value, metricDef) {
    const direction = metricDef.direction || 'higher-is-better';
    
    if (direction === 'lower-is-better') {
      if (value > metricDef.thresholds.critical) {
        return {
          name: 'critical',
          value: metricDef.thresholds.critical
        };
      } else if (value > metricDef.thresholds.warning) {
        return {
          name: 'warning',
          value: metricDef.thresholds.warning
        };
      }
    } else {
      if (value < metricDef.thresholds.critical) {
        return {
          name: 'critical',
          value: metricDef.thresholds.critical
        };
      } else if (value < metricDef.thresholds.warning) {
        return {
          name: 'warning',
          value: metricDef.thresholds.warning
        };
      }
    }
    
    return {
      name: 'target',
      value: metricDef.thresholds.target
    };
  }
  
  calculateAnomalySeverity(value, metricDef) {
    const direction = metricDef.direction || 'higher-is-better';
    
    if (direction === 'lower-is-better') {
      if (value > metricDef.thresholds.critical) {
        return 'critical';
      } else if (value > metricDef.thresholds.warning) {
        return 'warning';
      }
    } else {
      if (value < metricDef.thresholds.critical) {
        return 'critical';
      } else if (value < metricDef.thresholds.warning) {
        return 'warning';
      }
    }
    
    return 'normal';
  }
  
  calculateMean(values) {
    return values.reduce((sum, value) => sum + value, 0) / values.length;
  }
  
  calculateStandardDeviation(values, mean) {
    if (!mean) {
      mean = this.calculateMean(values);
    }
    
    const squareDiffs = values.map(value => Math.pow(value - mean, 2));
    const variance = squareDiffs.reduce((sum, value) => sum + value, 0) / values.length;
    return Math.sqrt(variance);
  }

  async calculatePassiveIncome() {
    try {
      // Get income data from data store
      const today = new Date();
      const yesterday = new Date(today);
      yesterday.setDate(yesterday.getDate() - 1);
      
      const weekStart = new Date(today);
      weekStart.setDate(weekStart.getDate() - weekStart.getDay());
      
      const monthStart = new Date(today.getFullYear(), today.getMonth(), 1);
      
      // Get income for different time periods
      const dailyIncome = await this.dataStore.getPassiveIncome({
        startDate: yesterday,
        endDate: today
      });
      
      const weeklyIncome = await this.dataStore.getPassiveIncome({
        startDate: weekStart,
        endDate: today
      });
      
      const monthlyIncome = await this.dataStore.getPassiveIncome({
        startDate: monthStart,
        endDate: today
      });
      
      // Calculate totals
      const calculateTotal = (incomeData) => {
        return incomeData.reduce((total, income) => total + income.amount, 0);
      };
      
      // Calculate income by source
      const incomeBySource = {};
      const allIncome = await this.dataStore.getPassiveIncome({
        startDate: new Date(today.getFullYear(), today.getMonth() - 1, today.getDate()),
        endDate: today
      });
      
      for (const income of allIncome) {
        if (!incomeBySource[income.source]) {
          incomeBySource[income.source] = {
            total: 0,
            count: 0
          };
        }
        
        incomeBySource[income.source].total += income.amount;
        incomeBySource[income.source].count++;
      }
      
      // Calculate income trends
      const previousMonthStart = new Date(today.getFullYear(), today.getMonth() - 1, 1);
      const previousMonthEnd = new Date(today.getFullYear(), today.getMonth(), 0);
      
      const previousMonthIncome = await this.dataStore.getPassiveIncome({
        startDate: previousMonthStart,
        endDate: previousMonthEnd
      });
      
      const previousMonthTotal = calculateTotal(previousMonthIncome);
      const currentMonthTotal = calculateTotal(monthlyIncome);
      
      let monthlyGrowth = 0;
      if (previousMonthTotal > 0) {
        monthlyGrowth = ((currentMonthTotal - previousMonthTotal) / previousMonthTotal) * 100;
      }
      
      // Calculate trends by source
      const trends = [];
      for (const [source, data] of Object.entries(incomeBySource)) {
        // Get previous month data for this source
        const previousSourceIncome = previousMonthIncome.filter(income => income.source === source);
        const previousSourceTotal = calculateTotal(previousSourceIncome);
        
        let trend = 0;
        if (previousSourceTotal > 0) {
          trend = ((data.total - previousSourceTotal) / previousSourceTotal) * 100;
        }
        
        trends.push({
          source,
          amount: data.total,
          count: data.count,
          trend
        });
      }
      
      // Sort trends by amount
      trends.sort((a, b) => b.amount - a.amount);
      
      return {
        daily: calculateTotal(dailyIncome),
        weekly: calculateTotal(weeklyIncome),
        monthly: calculateTotal(monthlyIncome),
        total: calculateTotal(allIncome),
        monthlyGrowth,
        trends
      };
    } catch (error) {
      this.logger.error('Error calculating passive income', error);
      return {
        daily: 0,
        weekly: 0,
        monthly: 0,
        total: 0,
        monthlyGrowth: 0,
        trends: []
      };
    }
  }
